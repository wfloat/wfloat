cmake_minimum_required(VERSION 3.18)
project(wfloat-next-native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(WFLOAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(WFLOAT_LLAMA_DIR "${WFLOAT_ROOT}/vendor/llama.cpp" CACHE PATH "Matching llama source tree")
set(WFLOAT_SHERPA_DIR "${WFLOAT_ROOT}/vendor/sherpa-onnx" CACHE PATH "Matching sherpa source tree")
option(WFLOAT_NEXT_INCLUDE_LEGACY_LLM "Include legacy wfloat-core LLM C ABI in the same runtime archive" OFF)
option(WFLOAT_NEXT_TESTS "Build native contract and parser tests" OFF)
option(WFLOAT_NEXT_BUILD_LLAMA "Build matching llama and common dependencies" ON)
if(WFLOAT_NEXT_BUILD_LLAMA AND NOT TARGET llama-common)
  set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
  foreach(name TESTS TOOLS EXAMPLES SERVER APP UI HTML)
    set(LLAMA_BUILD_${name} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
  set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  add_subdirectory("${WFLOAT_LLAMA_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
endif()
add_library(wfloat-next-runtime STATIC NextRuntime.cpp LlmRuntime.cpp SpeechRuntime.cpp
  "${WFLOAT_ROOT}/native/wfloat-core/src/schema/validator.cc")
if(WFLOAT_NEXT_INCLUDE_LEGACY_LLM)
  target_sources(wfloat-next-runtime PRIVATE "${WFLOAT_ROOT}/native/wfloat-core/src/wfloat_llm.cc")
  target_include_directories(wfloat-next-runtime PRIVATE "${WFLOAT_ROOT}/native/wfloat-core/include")
  target_compile_definitions(wfloat-next-runtime PRIVATE WFLOAT_ENABLE_LLAMA_CPP=1)
endif()
set_target_properties(wfloat-next-runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(wfloat-next-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE
  "${WFLOAT_LLAMA_DIR}/vendor" "${WFLOAT_SHERPA_DIR}" "${WFLOAT_ROOT}/native/wfloat-core/src")
target_compile_features(wfloat-next-runtime PUBLIC cxx_std_17)
target_link_libraries(wfloat-next-runtime PRIVATE llama llama-common)
# Platform owner must provide matching c-api AND csrc VAD implementation.
# Existing opaque C/C++ API dylibs alone may not export internal VadModel symbols.
if(TARGET sherpa-onnx-c-api AND TARGET sherpa-onnx-core)
  target_link_libraries(wfloat-next-runtime PRIVATE sherpa-onnx-c-api sherpa-onnx-core)
endif()
if(WFLOAT_NEXT_TESTS)
  enable_testing()
  add_executable(next-llm-model-params-test tests/llm_model_params_test.cpp)
  target_include_directories(next-llm-model-params-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(next-llm-model-params-test PRIVATE llama)
  add_test(NAME next-llm-model-params COMMAND next-llm-model-params-test)
  add_executable(next-llm-assets-test tests/llm_assets_test.cpp)
  target_include_directories(next-llm-assets-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${WFLOAT_LLAMA_DIR}/vendor")
  add_test(NAME next-llm-assets COMMAND next-llm-assets-test)
  find_package(Threads REQUIRED)
  add_executable(next-runtime-test tests/runtime_test.cpp NextRuntime.cpp)
  target_include_directories(next-runtime-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${WFLOAT_LLAMA_DIR}/vendor")
  target_link_libraries(next-runtime-test PRIVATE Threads::Threads)
  add_test(NAME next-runtime-contract COMMAND next-runtime-test)
  add_executable(next-speech-test tests/speech_test.cpp)
  target_include_directories(next-speech-test PRIVATE "${WFLOAT_LLAMA_DIR}/vendor" "${WFLOAT_SHERPA_DIR}")
  target_link_libraries(next-speech-test PRIVATE Threads::Threads)
  add_test(NAME next-speech-contract COMMAND next-speech-test)
  add_executable(next-llm-smoke tests/llm_smoke.cpp)
  target_include_directories(next-llm-smoke PRIVATE "${WFLOAT_LLAMA_DIR}/vendor")
  target_link_libraries(next-llm-smoke PRIVATE wfloat-next-runtime Threads::Threads)
  add_executable(next-parser-test tests/parser_test.cpp "${WFLOAT_ROOT}/native/wfloat-core/src/schema/validator.cc")
  target_include_directories(next-parser-test PRIVATE "${WFLOAT_ROOT}/native/wfloat-core/src")
  target_compile_definitions(next-parser-test PRIVATE WFLOAT_TEMPLATE_DIR="${WFLOAT_LLAMA_DIR}/models/templates" WFLOAT_SCHEMA_CASES="${WFLOAT_ROOT}/tests/schema/cases.json")
  target_link_libraries(next-parser-test PRIVATE llama llama-common)
  add_test(NAME next-parser COMMAND next-parser-test)
endif()
